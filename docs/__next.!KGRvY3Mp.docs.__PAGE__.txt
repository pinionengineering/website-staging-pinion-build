// sw-pub virtualizes every root into uniform super-blocks, so the setup
// document carries a block_count instead of a per-block CID list — no
// manifest of any kind is needed.
type rootInfo struct {
	BlockCount int `json:"block_count"`
}

type clientState struct {
	Server      string              `json:"server"`
	KeyID       string              `json:"key_id"`
	Protocol    string              `json:"protocol"`
	ClientSetup []byte              `json:"client_setup"`
	Roots       map[string]rootInfo `json:"roots"`
}

func main() {
	data, err := os.ReadFile("prover-state.json")
	if err != nil {
		log.Fatal(err)
	}
	var state clientState
	if err := json.Unmarshal(data, &state); err != nil {
		log.Fatal(err)
	}

	// Reconstruct each root's super-block challenge ids: rootCID.Bytes() ||
	// BE64(localIndex) for localIndex in [0, block_count). Both the server and
	// this client derive the same ids independently from the root CID plus
	// block_count alone — see ipfsproof.SuperBlockID.
	var blockIDs [][]byte
	for rootStr, root := range state.Roots {
		rootCID, err := cid.Decode(rootStr)
		if err != nil {
			log.Fatalf("invalid root CID %s: %v", rootStr, err)
		}
		for i := 0; i < root.BlockCount; i++ {
			blockIDs = append(blockIDs, ipfsproof.SuperBlockID(rootCID, uint64(i)))
		}
	}

	factory := swpub.NewChallengerFactory()
	challenger, err := factory.NewChallenger(state.ClientSetup, 20)
	if err != nil {
		log.Fatal(err)
	}

	for round := 1; ; round++ {
		// Step 1: generate challenge locally, no network call.
		chal, validator, err := challenger.Challenge(blockIDs)
		if err != nil {
			log.Fatalf("round %d: challenge failed: %v", round, err)
		}

		// Step 2: send challenge to the prover, no auth required.
		body, _ := json.Marshal(map[string]any{
			"key_id":    state.KeyID,
			"roots":     []string{},
			"challenge": chal,
		})
		resp, err := http.Post(
			state.Server+"/prover/prove",
			"application/json",
			bytes.NewReader(body),
		)
		if err != nil {
			log.Fatalf("round %d: prove request failed: %v", round, err)
		}
		body, _ = io.ReadAll(resp.Body)
		resp.Body.Close()
		var jr struct {
			JobID string `json:"job_id"`
		}
		if err := json.Unmarshal(body, &jr); err != nil {
			log.Fatalf("round %d: decode prove response: %v", round, err)
		}

		// Step 3: proving is asynchronous — poll until the job finishes.
		var proof []byte
		for {
			time.Sleep(500 * time.Millisecond)
			sresp, err := http.Get(state.Server + "/prover/prove/" + jr.JobID)
			if err != nil {
				log.Fatalf("round %d: poll failed: %v", round, err)
			}
			sbody, _ := io.ReadAll(sresp.Body)
			sresp.Body.Close()
			var status struct {
				Status string `json:"status"`
				Proof  []byte `json:"proof"`
				Error  string `json:"error"`
			}
			if err := json.Unmarshal(sbody, &status); err != nil {
				log.Fatalf("round %d: decode status: %v", round, err)
			}
			if status.Status == "prove-done" {
				proof = status.Proof
				break
			}
			if status.Status == "prove-failed" {
				log.Fatalf("round %d: prove job failed: %s", round, status.Error)
			}
		}

		// Step 4: verify the proof locally, no network call.
		ok, err := validator.Verify(chal, proof)
		if err != nil || !ok {
			log.Fatalf("round %d: VERIFICATION FAILED, data may be lost or tampered", round)
		}
		fmt.Printf("round %d: OK\n", round)

		time.Sleep(10 * time.Second)
	}
